## 6. 可视化示例

### 6.1 时空同一化可视化
```python
import matplotlib.pyplot as plt
import numpy as np

# 时空同一化可视化
t = np.linspace(0, 10, 100)
c = 3  # 光速（简化为3）
x = c * t

plt.figure(figsize=(10, 6))
plt.plot(t, x, label='时空同一化轨迹')
plt.xlabel('时间')
plt.ylabel('空间位置')
plt.title('时空同一化可视化')
plt.legend()
plt.grid(True)
plt.show()
```

### 6.2 螺旋时空可视化
```python
import matplotlib.pyplot as plt
import numpy as np
from mpl_toolkits.mplot3d import Axes3D

# 螺旋时空可视化
t = np.linspace(0, 10, 1000)
r = 1  # 螺旋半径
omega = 1  # 角速度
h = 0.5  # 轴向速度

x = r * np.cos(omega * t)
y = r * np.sin(omega * t)
z = h * t

fig = plt.figure(figsize=(10, 8))
ax = fig.add_subplot(111, projection='3d')
ax.plot(x, y, z, label='螺旋时空轨迹')
ax.set_xlabel('X')
ax.set_ylabel('Y')
ax.set_zlabel('Z')
ax.set_title('螺旋时空可视化')
ax.legend()
plt.show()
```

### 6.3 力场可视化
```python
import matplotlib.pyplot as plt
import numpy as np

# 电场可视化
def electric_field(x, y, q, x0, y0):
    r = np.sqrt((x - x0)**2 + (y - y0)**2)
    Ex = q * (x - x0) / r**3
    Ey = q * (y - y0) / r**3
    return Ex, Ey

x = np.linspace(-5, 5, 20)
y = np.linspace(-5, 5, 20)
X, Y = np.meshgrid(x, y)

# 正电荷
Ex1, Ey1 = electric_field(X, Y, 1, -2, 0)
# 负电荷
Ex2, Ey2 = electric_field(X, Y, -1, 2, 0)

Ex = Ex1 + Ex2
Ey = Ey1 + Ey2

plt.figure(figsize=(10, 8))
plt.streamplot(X, Y, Ex, Ey, density=1.5, linewidth=1, color='b')
plt.scatter([-2, 2], [0, 0], c=['r', 'b'], s=100)
plt.xlabel('X')
plt.ylabel('Y')
plt.title('电场可视化（电偶极子）')
plt.axis('equal')
plt.show()
```
