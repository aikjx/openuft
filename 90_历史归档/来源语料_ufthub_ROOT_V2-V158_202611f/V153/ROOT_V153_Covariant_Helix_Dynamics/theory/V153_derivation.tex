\section*{Units and basic helical kinematics}
We impose only $c=1$ and keep $\hbar$ and $G$ dimensionful.
\[\mathbf r(t)=(a\cos\omega t,a\sin\omega t,ut),\quad
 a^2\omega^2+u^2=1,\quad
 \kappa=a\omega^2,\quad\tau=u\omega,\quad\omega^2=\kappa^2+\tau^2.\]
\section*{Pure-gauge obstruction of a single smooth scalar phase}
Let $A_\mu=\partial_\mu\theta$. Then
\[F_{\mu\nu}=\partial_\mu\partial_\nu\theta-\partial_\nu\partial_\mu\theta=0.\]
If $h_{\mu\nu}=\partial_\mu\partial_\nu\theta$ and $\xi_\mu=\tfrac12\partial_\mu\theta$, then
\[h_{\mu\nu}=\partial_\mu\xi_\nu+\partial_\nu\xi_\mu,\qquad
 R^{(1)}_{\mu\nu\rho\sigma}=0.\]
This is conditional on this local smooth ansatz and is not a no-go theorem for composite or emergent gauge fields.

\section*{Gauge symmetry implies source conservation}
\[\delta_\lambda A_\mu=\partial_\mu\lambda
\implies \delta S_{\mathrm{int}}^{(1)}=\int d^4x\,\lambda\partial_\mu J^\mu.\]
\[\delta_\xi h_{\mu\nu}=2\partial_{(\mu}\xi_{\nu)}
\implies\delta S_{\mathrm{int}}^{(2)}=-\int d^4x\,\xi_\nu\partial_\mu T^{\mu\nu}.\]
\section*{Dimensional consistency}
\[[\hbar]=ML,\quad[G]=L/M,\quad[F_{\mu\nu}]=L^{-2},\quad[R]=L^{-2}.\]
\[S=\int d^4x\sqrt{-g}\left(\frac{R}{16\pi G}-\frac{\hbar}{4g_F^2}F_{\mu\nu}F^{\mu\nu}\right),\quad[S]=ML.\]
This is an established sum of actions, not a common microphysical derivation.
