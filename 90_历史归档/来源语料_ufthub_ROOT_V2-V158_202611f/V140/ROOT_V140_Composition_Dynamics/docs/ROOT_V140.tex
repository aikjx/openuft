\section{Conventions}
Only $c=1$; $\hbar$ and $G$ remain dimensionful. Signature $(-,+,+,+)$.
\section{Helix identity}
For $\mathbf r=(a\cos\omega t,a\sin\omega t,ut)$ and $a^2\omega^2+u^2=1$,
\begin{equation}\kappa=a\omega^2,\qquad \tau=u\omega,\qquad \kappa^2+\tau^2=\omega^2.\end{equation}
This is kinematics and does not imply a new force.
\section{Composite curvature}
\begin{equation}F=\sum_{a=1}^{2}dq_a\wedge dp_a=dA,\quad A=\sum_a q_a dp_a,\quad dF=0.\end{equation}
For a unit vector $\mathbf n=(\sin\theta\cos\phi,\sin\theta\sin\phi,\cos\theta)$,
$dq\wedge dp=\tfrac12\sin\theta d\theta\wedge d\phi$ locally with $q=(1-\cos\theta)/2$, $p=\phi$.
\begin{equation}\operatorname{Pf}(F)=F_{01}F_{23}-F_{02}F_{13}+F_{03}F_{12}=\det \frac{\partial(q_1,p_1,q_2,p_2)}{\partial(t,x,y,z)}=\mathbf E\cdot\mathbf B.\end{equation}
\section{Variational projection}
\begin{equation}S_F=-\frac{\hbar}{4g_F^2}\int d^4x\sqrt{-g}F_{\mu\nu}F^{\mu\nu}.\end{equation}
With $J^\nu=\nabla_\mu F^{\mu\nu}$, variations of the four scalars imply
\begin{equation}J^\nu\partial_\nu q_a=J^\nu\partial_\nu p_a=0,\quad a=1,2.\end{equation}
For independent gradients, $J=0$. For degenerate gradients, extra solutions may exist.
\section{Exact counterexample}
Let $q_1=z$, $p_1=x\sin(kz)-y\cos(kz)$, and $q_2=p_2=0$.
Then $\mathbf E=0$, $\mathbf B=(\cos kz,\sin kz,0)$ and
\begin{equation}J^\mu=(0,k\cos kz,k\sin kz,0)\ne0,\qquad J\cdot dq_a=J\cdot dp_a=0.\end{equation}
The projected Euler--Lagrange equations hold while vacuum Maxwell fails.
The force-free contraction $F_{\mu\nu}J^\nu=0$ implies stress-energy conservation in this example.
\section{Circularly polarized null wave}
Let $s=t-z$, $q_1=s$, $p_1=x\cos(ks)+y\sin(ks)$.
\begin{equation}\mathbf E=(\cos ks,\sin ks,0),\quad \mathbf B=(\sin ks,-\cos ks,0),\quad \mathbf E\cdot\mathbf B=0,\quad J^\mu=0.\end{equation}
The field direction rotates while propagation remains straight.
\section{Dimensional analysis}
$[\hbar]=ML$, $[G]=L/M$, $[F]=L^{-2}$, $[R]=L^{-2}$;
$[\int\hbar F^2 d^4x]=[\int R d^4x/G]=ML$.
