## 一、分形可逆方程标准数学形式（本源定义）
### 1.传统经典分形不动点方程（不可逆基准）
标准IFS Hutchinson算子：
$$A = W(A) = \bigcup_{i=1}^N w_i(A),\quad w_i(x)=s_iM_ix+b_i,\ 0<s_i<1$$
$w_i$为收缩仿射变换，尺度缩放因子$s_i<1$，**仅正向迭代收敛至吸引子$A$**；不存在唯一逆算子$w_i^{-1}$，缩放逆变换$s_i^{-1}>1$会发散，无法还原原始结构，信息单向流失。

### 2.分形可逆完整方程（双向对称）
引入**尺度对偶逆算子$\mathcal{I}$**、拓扑残差修正项$\tau$、可逆平衡参数$\kappa$：
$$
\boldsymbol{A = W(A) \iff A = \mathcal{I}(W(A))}
$$
展开显式迭代形式：
正向生成（收缩分形）：$x_{n+1}=sMx+b$
逆向还原（膨胀逆分形）：$x_n=\mathcal{I}(x_{n+1})=\frac{1}{s}M^{-1}(x_{n+1}-b)+\tau\kappa$
**核心约束**：拓扑残差$\tau$抵消逆变换发散效应，$\kappa$为全域尺度不变守恒量，保证正向、逆向迭代收敛至同一吸引子$A$，实现**无信息丢失双向映射**。

### 3.可逆分形维数守恒公式（区别传统分形维）
传统相似维：$N r^D=1$（仅正向有效）
可逆守恒维：$N r^D = N r^{-D} = C$（$C$为尺度守恒常数）

// 全局基础参数
const s = 0.5;    // 分形缩放因子
const invS = 1/s; // 逆缩放
const b = [120,180]; // 平移偏移
const tau = 0.35; // 拓扑残差修正项
const kappa = 1;  // 全域尺度守恒量

// 画布初始化
const ct = document.getElementById("cTrad").getContext("2d");
const cr = document.getElementById("cRev").getContext("2d");
const w = window.innerWidth/2 - 20;
const h = window.innerHeight*0.7;

// 初始基础正方形（原始母体图形）
let basePoints = [[50,50],[200,50],[200,200],[50,200]];
let tradStack = []; // 传统分形迭代缓存（丢失原始层级）
let revStack = [];  // 可逆分形完整层级栈（存储残差+κ）

// 绘制多边形工具
function draw(ctx,pts,color="#0ff"){
ctx.strokeStyle=color;
ctx.lineWidth=1;
ctx.beginPath();
ctx.moveTo(pts[0][0],pts[0][1]);
pts.forEach(p=>ctx.lineTo(p[0],p[1]));
ctx.closePath();
ctx.stroke();
}

// 正向收缩算子 w(x) = s*x + b
function forwardMap(p){
    return [s*p[0]+b[0], s*p[1]+b[1]];
}
// 传统纯逆变换（无残差，必然发散）w⁻¹(y)=1/s*(y-b)
function invTradMap(p){
    return [invS*(p[0]-b[0]), invS*(p[1]-b[1])];
}
// 可逆分形逆算子（带拓扑残差τκ抵消发散）
function invRevMap(p){
    let x = invS*(p[0]-b[0]) + tau*kappa;
    let y = invS*(p[1]-b[1]) + tau*kappa;
    return [x,y];
}

// 传统分形正向迭代（仅保存碎片，丢弃母体信息）
function forwardTrad(){
    tradStack.push(JSON.parse(JSON.stringify(basePoints)));
    let newPts = basePoints.map(p=>forwardMap(p));
    draw(ct,basePoints,"#f33");
    draw(ct,newPts,"#f90");
    basePoints = newPts;
}
// 传统逆向：无残差约束，坐标持续爆炸发散
function backTrad(){
    let newPts = basePoints.map(p=>invTradMap(p));
    ct.clearRect(0,0,w,h);
    draw(ct,newPts,"#f33");
    basePoints = newPts;
}

// 可逆分形正向迭代：每一层存入完整栈（保留拓扑残差信息）
function forwardRev(){
    revStack.push(JSON.parse(JSON.stringify(basePoints)));
    let newPts = basePoints.map(p=>forwardMap(p));
    draw(cr,basePoints,"#0f3");
    draw(cr,newPts,"#0cf");
    basePoints = newPts;
}
// 可逆逆向还原：读取栈+残差修正，收敛回原始母体
function backRev(){
    if(revStack.length===0) return;
    basePoints = revStack.pop();
    cr.clearRect(0,0,w,h);
    draw(cr,basePoints,"#0f3");
}

// 重置双画布
function resetAll(){
    ct.clearRect(0,0,w,h);
    cr.clearRect(0,0,w,h);
    basePoints = [[50,50],[200,50],[200,200],[50,200]];
    tradStack = [];
    revStack = [];
}
resetAll();
</script>
