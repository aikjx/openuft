$$
\hat H_{\text{red}}=\sum_{\mathbf{k}\sigma}\varepsilon_{\mathbf{k}}c^\dagger_{\mathbf{k}\sigma}c_{\mathbf{k}\sigma}-|V|\sum_{\mathbf{kk}'}c^\dagger_{\mathbf{k}\uparrow}c^\dagger_{-\mathbf{k}\downarrow}c_{-\mathbf{k}'\downarrow}c_{\mathbf{k}'\uparrow},\qquad |\varepsilon_{\mathbf{k}}|<\hbar\omega_D
