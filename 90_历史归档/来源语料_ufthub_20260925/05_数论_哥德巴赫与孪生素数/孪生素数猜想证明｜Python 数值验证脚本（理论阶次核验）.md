def is_prime(num):
    if num < 2:
        return False
    if num == 2:
        return True
    if num % 2 == 0:
        return False
    max_sqrt = int(math.isqrt(num)) + 1
    for i in range(3, max_sqrt, 2):

            return False
    return True
