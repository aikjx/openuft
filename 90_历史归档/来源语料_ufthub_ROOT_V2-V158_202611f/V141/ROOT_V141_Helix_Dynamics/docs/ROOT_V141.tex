\section{Field-redefinition obstruction}
Let $\Omega=\sum_a dq_a\wedge dp_a=d\beta$, $\beta=\sum_aq_a dp_a$. Then
\begin{equation}
\mathcal H=dA+\eta\Omega=d\left(A+\eta\beta\right)=dA_{\rm eff}.
\end{equation}
When no other term distinguishes $A$ and $\beta$, the geometric contribution is locally absorbable.
\section{Dynamical action}
In signature $(-,+,+,+)$, take a dimensionless scalar $\theta$ and an independent gauge potential $A_\mu$:
\begin{equation}
S=\hbar\int d^4x\sqrt{-g}\left[\frac{R}{16\pi G\hbar}
-\frac{F_{\mu\nu}F^{\mu\nu}}{4g_F^2}
-\frac{f^2}{2}\nabla_\mu\theta\nabla^\mu\theta
-\frac{\lambda\theta}{4}F_{\mu\nu}\widetilde F^{\mu\nu}\right].
\end{equation}
$[A_\mu]=L^{-1}$, $[F_{\mu\nu}]=L^{-2}$, $[f]=L^{-1}$,
$[G]=L/M$, $[\hbar]=ML$, hence $[G\hbar]=L^2$.
The independent variations yield
\begin{align}
\nabla_\mu\left(g_F^{-2}F^{\mu\nu}+\lambda\theta\widetilde F^{\mu\nu}\right)&=0,\\
f^2\Box\theta&=\frac{\lambda}{4}F_{\mu\nu}\widetilde F^{\mu\nu}.
\end{align}
In the flat constant-$\theta$ limit this recovers the vacuum Maxwell equations with two photon polarizations.
\section{One-mode reduction}
Let $\mathbf A(t,z)=a(t)(\cos kz,-\sin kz,0)$, $A_0=0$, and $\theta=\theta(t)$. Then $\nabla\times\mathbf A=k\mathbf A$, $\mathbf E=-\dot a\mathbf e$, $\mathbf B=ka\mathbf e$. Hence
\begin{equation}
\frac{\mathcal L_{\rm red}}{\hbar}=\frac{\dot a^2-k^2a^2}{2g_F^2}
+\frac{f^2\dot\theta^2}{2}-\lambda k\theta a\dot a.
\end{equation}
The Euler--Lagrange equations are
\begin{align}
\ddot a+k^2 a-g_F^2\lambda k\dot\theta a&=0,\\
f^2\ddot\theta+\lambda k a\dot a&=0.
\end{align}
Two first integrals are
\begin{align}
\frac{\mathcal E}{\hbar}&=\frac{\dot a^2+k^2a^2}{2g_F^2}+\frac{f^2\dot\theta^2}{2},\\
Q&=f^2\dot\theta+\frac{\lambda k}{2}a^2.
\end{align}
Both derivatives vanish upon substituting the field equations. The topological coupling does not add an explicit energy-density term.
