\section{Action and monopole flux}
\begin{align}
S_6&=\int d^6X\sqrt{-g_6}\left[\frac{K_6}{2}{\cal R}_6-\Lambda_6-\frac{\chi_6}{4}F_{MN}F^{MN}\right],\\

\section{Radion reduction and stationary points}
\begin{align}
K_4&=4\pi K_6R_0^2=\frac1{8\pi G_4},\\
V_E(R)&=\left(\frac{R_0}{R}\right)^4 \left[4\pi\Lambda_6R^2-4\pi K_6+\frac{\pi\chi_6n^2}{2R^2}\right]
=\frac A{R^2}-\frac B{R^4}+\frac C{R^6},\\
A&=4\pi\Lambda_6R_0^4,\quad B=4\pi K_6R_0^4,\quad C=\frac{\pi\chi_6n^2R_0^4}{2},\\
\varphi&=2\sqrt{K_4}\ln(R/R_0),\qquad
m_\varphi^2=\frac{R_*^2}{4K_4}\left.\frac{d^2V}{dR^2}\right|_{R_*}.
\end{align}
For $y=R^2$, $V(y)=A/y-B/y^2+C/y^3$, whence
\begin{align}
-Ay^2+2By-3C&=0,\\
y_\pm&=\frac{B\pm\sqrt{B^2-3AC}}A,\qquad
B^2>3AC\text{ is required for strict radial extrema},\\
V(y_*)=0\ \land\ V'(y_*)&=0\quad\Longleftrightarrow\quad B^2=4AC,
\quad y_*=\frac{B}{2A},\\
\Lambda_6\chi_6n^2&=2K_6^2.
\end{align}

0&=-\Lambda_6+\frac{\chi_6n^2}{8R^4}.
